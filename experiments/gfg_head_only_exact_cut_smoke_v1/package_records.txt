def sha(p):
    h=hashlib.sha256()
    with p.open('rb') as f:
        for b in iter(lambda:f.read(4*1024*1024),b''):h.update(b)
    return h.hexdigest()
def write(p,o):
    with p.open('x',encoding='utf8') as f:json.dump(o,f,indent=2,allow_nan=False)
